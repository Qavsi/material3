cmake_minimum_required(VERSION 3.30)
project(QcmColorReference LANGUAGES CXX)
find_package(Qt6 6.12 REQUIRED COMPONENTS Quick)
qt_standard_project_setup(REQUIRES 6.12)
set(QCM_ROOT "" CACHE PATH "QmlMaterial reference checkout with a compiled .qavsi-build")
if(NOT EXISTS "${QCM_ROOT}/include/qml_material/token/color.hpp")
    message(FATAL_ERROR "Pass -DQCM_ROOT=<QmlMaterial checkout>")
endif()
find_library(QCM_COLOR_REFERENCE_LIBRARY NAMES qml_material
    HINTS "${QCM_ROOT}/.qavsi-build" NO_DEFAULT_PATH REQUIRED)
qt_add_executable(qcm-color-reference main.cpp)
target_compile_features(qcm-color-reference PRIVATE cxx_std_20)
target_include_directories(qcm-color-reference PRIVATE
    "${QCM_ROOT}/include" "${QCM_ROOT}/third_party/material-color")
target_link_libraries(qcm-color-reference PRIVATE Qt6::Quick "${QCM_COLOR_REFERENCE_LIBRARY}")
