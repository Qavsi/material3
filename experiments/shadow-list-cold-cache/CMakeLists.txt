cmake_minimum_required(VERSION 3.30)
project(QavsiShadowListStudy LANGUAGES CXX)
find_package(Qt6 6.12 REQUIRED COMPONENTS Quick Test GuiPrivate LabsStyleKit)
qt_standard_project_setup(REQUIRES 6.12)
set(QT_QML_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qml")
set(QAVSI_MATERIAL3_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(QAVSI_MATERIAL3_BUILD_COMPARISON OFF CACHE BOOL "" FORCE)
set(QAVSI_MATERIAL3_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(QAVSI_ICONS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(QAVSI_ICONS_BUILD_MATERIAL OFF CACHE BOOL "" FORCE)
add_subdirectory("${STUDY_SNAPSHOT}/material3" material3)
qt_add_executable(shadow-list-study MACOSX_BUNDLE main.cpp)
target_compile_features(shadow-list-study PRIVATE cxx_std_20)
target_link_libraries(shadow-list-study PRIVATE Qt6::Quick Qt6::Test Qt6::GuiPrivate QavsiMaterial3::Style)
target_compile_definitions(shadow-list-study PRIVATE STUDY_QML_DIR="${CMAKE_CURRENT_SOURCE_DIR}" STUDY_IMPORT_DIR="${QT_QML_OUTPUT_DIRECTORY}")
set_target_properties(shadow-list-study PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER dev.qavsi.shadow-list-study)
add_library(drawcounter SHARED drawcounter.cpp)
target_compile_features(drawcounter PRIVATE cxx_std_20)
target_link_libraries(drawcounter PRIVATE Qt6::GuiPrivate)
