cmake_minimum_required(VERSION 3.30)
project(QavsiShadowBenchmark LANGUAGES CXX)
find_package(Qt6 6.12 REQUIRED COMPONENTS Quick Test GuiPrivate)
qt_standard_project_setup(REQUIRES 6.12)
set(QM_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(QM_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(QM_USE_PCH ON CACHE BOOL "" FORCE)
set(QML_MATERIAL_BUILD_TYPE SHARED CACHE STRING "" FORCE)
add_subdirectory("${QCM_SOURCE}" qcm)
function(add_benchmark target)
  qt_add_executable(${target} MACOSX_BUNDLE main.cpp)
  target_compile_features(${target} PRIVATE cxx_std_20)
  target_link_libraries(${target} PRIVATE Qt6::Quick Qt6::Test Qt6::GuiPrivate qml_material)
  target_compile_definitions(${target} PRIVATE BENCH_QML_DIR="${CMAKE_CURRENT_SOURCE_DIR}" QCM_QML_DIR="${CMAKE_BINARY_DIR}/qml_modules")
  set_target_properties(${target} PROPERTIES MACOSX_BUNDLE_GUI_IDENTIFIER "dev.qavsi.${target}")
endfunction()
add_benchmark(shadow-benchmark)
# Development-only profiler/preview executable; not used for reported timing runs.
add_benchmark(shadow-preview)
target_compile_definitions(shadow-preview PRIVATE QT_QML_DEBUG)
