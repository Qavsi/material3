cmake_minimum_required(VERSION 3.30)
project(QavsiShadowPortBenchmark LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 6.12 REQUIRED COMPONENTS Quick Test GuiPrivate)
qt_standard_project_setup(REQUIRES 6.12)
set(QT_QML_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/qml_modules")
set(QAVSI_MATERIAL3_BUILD_EXAMPLE OFF CACHE BOOL "" FORCE)
set(QAVSI_MATERIAL3_BUILD_COMPARISON OFF CACHE BOOL "" FORCE)
set(QAVSI_MATERIAL3_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(QAVSI_ICONS_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(QAVSI_ICONS_BUILD_MATERIAL OFF CACHE BOOL "" FORCE)
if(NOT QAVSI_MATERIAL3_SOURCE)
    message(FATAL_ERROR "QAVSI_MATERIAL3_SOURCE must point to the actual ported Material3 kit")
endif()
if(NOT QCM_REFERENCE_BUILD)
    message(FATAL_ERROR "QCM_REFERENCE_BUILD must point to the existing frozen Release QCM build")
endif()
set(qcm_library "${QCM_REFERENCE_BUILD}/qcm/libqml_material.dylib")
if(NOT EXISTS "${qcm_library}")
    message(FATAL_ERROR "Existing QCM reference dylib not found: ${qcm_library}")
endif()
add_library(QcmReference SHARED IMPORTED)
set_target_properties(QcmReference PROPERTIES IMPORTED_LOCATION "${qcm_library}")
add_subdirectory("${QAVSI_MATERIAL3_SOURCE}" material3)
qt_add_executable(shadow-port-benchmark MACOSX_BUNDLE main.cpp)
target_compile_features(shadow-port-benchmark PRIVATE cxx_std_20)
target_link_libraries(shadow-port-benchmark PRIVATE
    Qt6::Quick Qt6::Test Qt6::GuiPrivate QcmReference QavsiMaterial3::Style)
target_compile_definitions(shadow-port-benchmark PRIVATE
    BENCH_QML_DIR="${CMAKE_CURRENT_SOURCE_DIR}"
    QCM_QML_DIR="${QCM_REFERENCE_BUILD}/qml_modules"
    M3_QML_DIR="${QT_QML_OUTPUT_DIRECTORY}")
set_target_properties(shadow-port-benchmark PROPERTIES
    MACOSX_BUNDLE_GUI_IDENTIFIER "dev.qavsi.shadow-port-benchmark")
